Project Description

A simple and clean to-do list web application built with PHP, MySQL, and the MVC architectural pattern. Users can register, log in, and manage personal tasks with session-based authentication.

The project uses GitHub Issues for planning and tracking progress, and follows a clear MVC file organization for separation of concerns between models, views, and controllers.

Key Features

  • User registration and login
  • Session-based authentication
  • Cookie tracks total number of login visits
  • Full CRUD task management — Create, Read, Update, Delete tasks
  • MVC file organization for clear separation of concerns
  • GitHub Issues used for planning and tracking progress

Sample Code — Session-Based Auth (PHP)

<?php
// login.php — verify credentials and start session
session_start();

$conn = new mysqli('localhost', 'user', 'pass', 'todo_db');

$email    = $conn->real_escape_string($_POST['email']);
$password = $_POST['password'];

$result = $conn->query("SELECT * FROM users WHERE email='$email' LIMIT 1");
$user   = $result->fetch_assoc();

if ($user && password_verify($password, $user['password_hash'])) {
    $_SESSION['user_id'] = $user['id'];
    $_SESSION['email']   = $user['email'];

    // Cookie: track total login visits
    $visits = isset($_COOKIE['login_visits']) ? (int)$_COOKIE['login_visits'] + 1 : 1;
    setcookie('login_visits', $visits, time() + (86400 * 365));

    header('Location: tasks.php');
} else {
    header('Location: login.php?error=invalid');
}
$conn->close();
?>

Sample Code — CRUD Task Query (PHP / MySQL)

<?php
// TaskModel.php — MVC Model layer
class TaskModel {
    private $conn;

    public function __construct($conn) {
        $this->conn = $conn;
    }

    public function getTasksByUser($userId) {
        $id     = intval($userId);
        $result = $this->conn->query(
            "SELECT * FROM tasks WHERE user_id=$id ORDER BY created_at DESC"
        );
        return $result->fetch_all(MYSQLI_ASSOC);
    }

    public function createTask($userId, $title) {
        $id    = intval($userId);
        $title = $this->conn->real_escape_string($title);
        $this->conn->query(
            "INSERT INTO tasks (user_id, title) VALUES ($id, '$title')"
        );
        return $this->conn->insert_id;
    }

    public function deleteTask($taskId, $userId) {
        $tid = intval($taskId);
        $uid = intval($userId);
        $this->conn->query(
            "DELETE FROM tasks WHERE id=$tid AND user_id=$uid"
        );
    }
}
?>
