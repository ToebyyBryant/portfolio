Sample Code — PHP / MySQL Connection & Query

<?php
// db_connect.php — replace credentials with your own
$host = 'localhost';
$db   = 'your_database';
$user = 'your_username';
$pass = 'your_password';

$conn = new mysqli($host, $user, $pass, $db);
if ($conn->connect_error) {
    die('Connection failed: ' . $conn->connect_error);
}

// Example: fetch all records from a table
$result = $conn->query("SELECT * FROM your_table ORDER BY id DESC");
while ($row = $result->fetch_assoc()) {
    echo "<p>" . htmlspecialchars($row['column_name']) . "</p>";
}
$conn->close();
?>
